# =============================================================================
# requirements.txt — SEC EDGAR Financial RAG Pipeline
# =============================================================================
#
# PINNING POLICY
# --------------
# All versions are pinned with >= minimum + < next breaking major so pip's
# resolver can satisfy transitive constraints without silent upgrades that
# break the pipeline.  Run `pip install -r requirements.txt` inside a fresh
# virtual-env (Python 3.10 or 3.11 recommended).
#
# GROUP ORDER mirrors the pipeline execution order so the file doubles as
# architecture documentation.  Each group is self-contained: you can comment
# out a whole group if you are not running that phase locally.
#
# TO CREATE A LOCKED FILE (recommended for CI / Docker):
#   pip install pip-tools
#   pip-compile requirements.txt -o requirements.lock
# =============================================================================

# -----------------------------------------------------------------------------
# [1] INGESTION — downloading raw filings from SEC EDGAR
# -----------------------------------------------------------------------------

# Requests — synchronous HTTP client for EDGAR API calls.
# We use synchronous requests (not httpx) here because EDGAR requires sequential
# rate-limiting (10 req/s) which is simpler to enforce in a sync loop.
requests>=2.31,<3.0

# httpx — async HTTP client used in the FastAPI service layer (async endpoints).
httpx>=0.25,<1.0

# aiohttp — async HTTP alternative; used in ingestion when parallel ticker
# downloads are enabled (with asyncio.Semaphore for rate-limit enforcement).
aiohttp>=3.9,<4.0

# aiofiles — async file I/O for writing raw HTML files without blocking the
# event loop during parallel ingestion runs.
aiofiles>=23.0,<24.0


# -----------------------------------------------------------------------------
# [2] PARSING — converting messy SEC HTML into structured JSON
# -----------------------------------------------------------------------------

# BeautifulSoup4 — HTML/XML parsing library; primary parser for SEC filings.
beautifulsoup4>=4.12,<5.0

# lxml — fast C-based XML/HTML parser used as the BeautifulSoup backend.
# Much more lenient with malformed HTML than html.parser; required for SEC filings
# which frequently have unclosed tags and stray XBRL inline elements.
lxml>=4.9,<6.0

# html5lib — fallback parser for the most severely malformed filings that lxml
# still cannot handle.  Slower but more spec-compliant.
html5lib>=1.1,<2.0

# pdfplumber — used only when the pipeline is pointed at PDF versions of filings
# (optional; the primary path is HTML).  Extracts text and tables from PDFs.
pdfplumber>=0.10,<1.0

# unstructured — document parsing library that handles mixed HTML/PDF/DOCX;
# used as an alternative/complement to BeautifulSoup for complex layouts.
# Pin the extras to avoid pulling in unnecessary heavy dependencies.
unstructured>=0.10,<1.0


# -----------------------------------------------------------------------------
# [3] CHUNKING — splitting parsed documents into indexable chunks
# -----------------------------------------------------------------------------

# sentence-transformers — loads all-MiniLM-L6-v2 for the SemanticChunker's
# sentence-level cosine-similarity boundary detection.  Also loads BGE-large

# Single package covers chunking + embedding + reranking for all local models.

# numpy — array math used in SemanticChunker for cosine similarity computation
# over sentence embedding matrices.

# scipy — used in SemanticChunker for cdist (pairwise cosine distances) and
# in the latency report for p50/p95/p99 percentile calculations.

# scikit-learn — used for cosine_similarity utility in semantic cache comparison
